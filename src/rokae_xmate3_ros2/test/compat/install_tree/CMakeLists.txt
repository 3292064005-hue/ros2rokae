cmake_minimum_required(VERSION 3.16)

if(NOT TARGET xCoreSDK::xCoreSDK_ros_bridge)
  message(FATAL_ERROR "xCoreSDK::xCoreSDK_ros_bridge target is missing from the install package")
endif()
if(NOT TARGET xCoreSDK::xCoreSDK_shared)
  message(FATAL_ERROR "xCoreSDK::xCoreSDK_shared target is missing from the install package")
endif()

  message(FATAL_ERROR "xCoreSDK::xCoreSDK_static target is missing from the install package")
endif()

  message(FATAL_ERROR "xCoreSDK_PRIMARY_INSTALL_CONSUMER must be exported as cxx_sdk_core_consumer")
endif()
if(NOT DEFINED xCoreSDK_STRICT_PUBLIC_INSTALL)
  message(FATAL_ERROR "xCoreSDK_STRICT_PUBLIC_INSTALL must be exported by the install-facing package")
endif()
if(NOT xCoreSDK_HAS_NATIVE_STATIC_TARGET)
  message(FATAL_ERROR "xCoreSDK::xCoreSDK_static must be a native static target in the install-facing package")
endif()
if(NOT DEFINED xCoreSDK_TARGET_FAMILY OR NOT xCoreSDK_TARGET_FAMILY STREQUAL "xmate_er3")
  message(FATAL_ERROR "xCoreSDK_TARGET_FAMILY must be exported as xmate_er3")
endif()
if(NOT DEFINED xCoreSDK_XMATE_ER3_ONLY OR NOT xCoreSDK_XMATE_ER3_ONLY)
  message(FATAL_ERROR "xCoreSDK_XMATE_ER3_ONLY must be TRUE for the install-facing package")
endif()
if(NOT DEFINED xCoreSDK_INTERNAL_SURFACE_ENABLED OR xCoreSDK_INTERNAL_SURFACE_ENABLED)
  message(FATAL_ERROR "xCoreSDK_INTERNAL_SURFACE_ENABLED must be FALSE for the install-facing package")
endif()
if(NOT DEFINED xCoreSDK_DEFAULT_COMPATIBILITY_ALIAS_POLICY OR NOT xCoreSDK_DEFAULT_COMPATIBILITY_ALIAS_POLICY STREQUAL "canonical_only")
  message(FATAL_ERROR "xCoreSDK_DEFAULT_COMPATIBILITY_ALIAS_POLICY must be exported as canonical_only")
endif()
if(NOT DEFINED xCoreSDK_PUBLIC_ROSIDL_ROOT OR NOT xCoreSDK_PUBLIC_ROSIDL_ROOT STREQUAL "srv/")
  message(FATAL_ERROR "xCoreSDK_PUBLIC_ROSIDL_ROOT must be exported as srv/")
endif()
if(NOT DEFINED xCoreSDK_INTERNAL_ROSIDL_ROOT OR NOT xCoreSDK_INTERNAL_ROSIDL_ROOT STREQUAL "internal_interfaces/srv/")
  message(FATAL_ERROR "xCoreSDK_INTERNAL_ROSIDL_ROOT must be exported as internal_interfaces/srv/")
endif()
if(TARGET xCoreSDK::rokae_xmate3_ros2_sdk_backend)
  message(FATAL_ERROR "private backend target must not leak into the install-facing package")
endif()
if(NOT xCoreSDK_STATIC_PROVIDER STREQUAL "native-static")
  message(FATAL_ERROR "xCoreSDK_STATIC_PROVIDER must report native-static")
endif()

set(XCORESDK_PRIMARY_INSTALL_LINK_TARGET "")
if(xCoreSDK_PRIMARY_INSTALL_CONSUMER STREQUAL "cxx_sdk_core_consumer")
  set(XCORESDK_PRIMARY_INSTALL_LINK_TARGET xCoreSDK::xCoreSDK_core)
else()
  message(FATAL_ERROR "unsupported xCoreSDK_PRIMARY_INSTALL_CONSUMER value: ${xCoreSDK_PRIMARY_INSTALL_CONSUMER}")
endif()

set(XCORESDK_CORE_ONLY_LINK_TARGET xCoreSDK::xCoreSDK_core)
set(XCORESDK_RUNTIME_LINK_TARGET xCoreSDK::xCoreSDK_shared)
if(NOT DEFINED XCORESDK_PRIMARY_INSTALL_LINK_TARGET OR NOT TARGET ${XCORESDK_PRIMARY_INSTALL_LINK_TARGET})
  message(FATAL_ERROR "primary install-facing target must resolve from xCoreSDK_PRIMARY_INSTALL_CONSUMER to an exported xCoreSDK target")
endif()
if(NOT TARGET ${XCORESDK_RUNTIME_LINK_TARGET})
  message(FATAL_ERROR "explicit runtime bridge target must resolve to an exported xCoreSDK runtime target")
endif()
if(NOT TARGET ${XCORESDK_CORE_ONLY_LINK_TARGET})
  message(FATAL_ERROR "xCoreSDK::xCoreSDK_core target is missing from the install package")
endif()

get_target_property(_xcore_static_includes xCoreSDK::xCoreSDK_static INTERFACE_INCLUDE_DIRECTORIES)
if(NOT _xcore_static_includes)
  message(FATAL_ERROR "xCoreSDK::xCoreSDK_static must publish install-tree include directories")
endif()
foreach(_inc IN LISTS _xcore_static_includes)
  if(_inc MATCHES "BUILD_INTERFACE" OR _inc MATCHES "CMAKE_CURRENT_SOURCE_DIR" OR _inc MATCHES "CMAKE_CURRENT_BINARY_DIR")
    message(FATAL_ERROR "install-facing static target leaked a build/source include path: ${_inc}")
  endif()
endforeach()

get_target_property(_xcore_primary_links ${XCORESDK_PRIMARY_INSTALL_LINK_TARGET} INTERFACE_LINK_LIBRARIES)
if(NOT _xcore_primary_links)
  message(FATAL_ERROR "primary install-facing target must link through exported install-compatible targets")
endif()
foreach(_link IN LISTS _xcore_primary_links)
  if(_link MATCHES "BUILD_INTERFACE" OR _link MATCHES "CMAKE_CURRENT_SOURCE_DIR" OR _link MATCHES "CMAKE_CURRENT_BINARY_DIR")
    message(FATAL_ERROR "primary runtime install-facing target leaked a build/source link dependency: ${_link}")
  endif()
endforeach()
